<script>
(function() {
    const searchInput = document.getElementById("search-input");
    const searchCounter = document.getElementById("search-counter");
    const grid = document.getElementById("org-grid");
    const letteraBtns = document.querySelectorAll(".lettera-btn");

    if (!grid) return;

    const cards = grid.querySelectorAll(".org-card");

    function filtra() {
        const query = searchInput.value.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "").replace(/[^\p{L}\p{N}]+/gu, " ").replace(/\s+/g, " ").trim();
        const letteraAttiva = document.querySelector(".lettera-btn--active");
        const lettera = letteraAttiva ? letteraAttiva.dataset.lettera : "all";
        let visibili = 0;

        cards.forEach(card => {
            const nome = card.dataset.cerca || "";
            const cardLettera = card.dataset.lettera;
            const matchLettera = (lettera === "all" || cardLettera === lettera);
            const matchRicerca = nome.includes(query);
            const visibile = matchLettera && matchRicerca;

            card.style.display = visibile ? "" : "none";

            if (visibile) visibili++;
        });

        if (searchCounter) {
            searchCounter.textContent = visibili + (visibili === 1 ? " organizzazione" : " organizzazioni");
        }
        let vuoto = document.getElementById("search-vuoto");
        if (!vuoto) {
            vuoto = document.createElement("p");
            vuoto.id = "search-vuoto";
            vuoto.className = "ami-vuoto";
            grid.insertAdjacentElement("afterend", vuoto);
        }
        vuoto.hidden = visibili > 0;
        if (!visibili) {
            vuoto.textContent = query ? "Nessuna organizzazione trovata per «" + searchInput.value.trim() + "»." : "Nessuna organizzazione con questa iniziale.";
        }
    }

    searchInput.addEventListener("input", filtra);

    letteraBtns.forEach(btn => {
        btn.addEventListener("click", function() {
            if (this.disabled) return;

            letteraBtns.forEach(b => {
                b.classList.remove("lettera-btn--active");
                if (!b.disabled) b.setAttribute("aria-pressed", "false");
            });
            this.classList.add("lettera-btn--active");
            this.setAttribute("aria-pressed", "true");
            filtra();
        });
    });

    filtra();
})();
</script>
